# Ranked bot backfill — Blood in the Sand server (bits-ranked-bots.md)
# Paste into Render → your server service → Environment → "Add from .env".
# These are the shipped defaults, written out explicitly so they're visible
# and one-click editable in the dashboard.

# Kill switch: 0 / off / false disables bot backfill AND the queue-size fuzz
# (honest numbers return automatically). Anything else = on.
RANKED_BOT_BACKFILL=on

# Empty-queue wait before a lone ranked queuer draws a bot match.
# Each queue entry gets a random deadline between MIN and MAX (jitter — an
# always-exactly-20s pop is itself a tell). Raise these as the real
# population grows, then flip the kill switch.

RANKED_BOT_MIN_WAIT_MS=15000
RANKED_BOT_MAX_WAIT_MS=25000
# The bot's advertised rating sits within ± this of the player's rating.
RANKED_BOT_RATING_JITTER=50
